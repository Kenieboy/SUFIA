// FILTER BY DATE

SELECT 
    pd.DATEDELIVERED,
    pd.PURCHASEDELIVERYNO,
    i.CODE AS ITEMCODE,
    i.NAMEENG AS MATERIALNAME,
    pdd.QTY,
    iu.DESCRIPTIONEN AS UNIT
FROM  PURCHASEDELIVERYDETAIL pdd
LEFT JOIN PURCHASEDELIVERY pd ON pd.ID = pdd.PURCHASEDELIVERYID
LEFT JOIN ITEMVARIATION iv ON pdd.ITEMVARIATIONID = iv.ID
LEFT JOIN  ITEM i ON iv.ITEMID = i.ID
LEFT JOIN  ITEMUNIT iu ON iv.ITEMUNITID = iu.ID
WHERE pd.DATEDELIVERED BETWEEN '2024-1-11' AND '2024-6-30'
ORDER BY pd.DATEDELIVERED DESC


// FILTER BY DATE REPORT FOR DAILYCONSUMPTION

SELECT 
    dc.PRODUCTITEMID AS ProductID,
    i.NAMEENG AS ProductName,
    ic.ID AS ItemCategoryID,
    ic.DESCRIPTION AS ItemCategoryDescription,
    iv.ID AS ItemVariationID,
    i2.NAMEENG AS ItemName,
    SUM(dcd.ACTUALQTY) AS Qty,  
    iu.DESCRIPTIONEN AS Unit,
    iv.COST AS UnitCost,
    SUM(dcd.ACTUALQTY * iv.COST) AS TotalCost 
FROM DAILYCONSUMPTION dc
JOIN DAILYCONSUMPTIONDETAIL dcd ON dc.ID = dcd.DAILYCONSUMPTIONID
JOIN ITEMVARIATION iv ON dcd.ITEMVARIATIONID = iv.ID
JOIN ITEM i2 ON iv.ITEMID = i2.ID
JOIN ITEM i ON dc.PRODUCTITEMID = i.ID
JOIN ITEMUNIT iu ON iv.ITEMUNITID = iu.ID
JOIN ITEMCATEGORY ic ON i.ITEMCATEGORYID = ic.ID  
WHERE dc.DATE BETWEEN '2025-01-01' AND '2025-02-1'
GROUP BY dc.PRODUCTITEMID, ic.ID, ic.DESCRIPTION, iv.ID, i2.NAMEENG, iu.DESCRIPTIONEN, iv.COST
ORDER BY dc.PRODUCTITEMID, ic.ID;
